extends SceneTree
func _initialize() -> void: call_deferred("run")
func rect(control: Control) -> Dictionary:
	var box := control.get_global_rect()
	return {"x":box.position.x,"y":box.position.y,"width":box.size.x,"height":box.size.y}
func run() -> void:
	Catalog.load_all()
	root.mode = Window.MODE_WINDOWED
	DisplayServer.window_move_to_foreground()
	DirAccess.make_dir_recursive_absolute("/tmp/pacific-density-before")
	root.content_scale_mode = Window.CONTENT_SCALE_MODE_DISABLED
	var app: Node = load("res://game/client_app.gd").new()
	app.capture_path = "user://unused-density.png"
	app.capture_frame = 1000000000
	root.add_child(app)
	await process_frame
	var folder := "user://tests/ui-density-"+RaceSetup.uuid()
	app.data = AppData.new(folder,true)
	app.checkpoint = RaceCheckpoint.new(app.data)
	app.menu.settings = app.data.settings
	app.data.profile.first_run_complete = true
	app.data.settings.reduce_motion = true
	app.data.mobile = true
	app.menu.mobile_override = true
	var rows: Array[Dictionary] = []
	for viewport: Vector2i in [Vector2i(884,398),Vector2i(960,720)]:
		root.size = viewport
		app.menu.logical_unit_override = 1.0 if viewport.y<500 else 1.5
		for scale: int in [100,150]:
			app.data.settings.ui_scale = scale
			for index in [30,39]:
				app.events.mode = "quick"
				app.events.filter_index = 0
				app.events.track_index = index
				app.events.show_event()
				for frame in range(8): await process_frame
				await RenderingServer.frame_post_draw
				assert(root.get_texture().get_image().save_png("/tmp/pacific-density-before/%dx%d-%d-%s.png"%[viewport.x,viewport.y,scale,Catalog.tracks[index].id])==OK)
				var button: Button = app.events.buttons[index]
				var font: Font = button.get_theme_font("font")
				var font_size := button.get_theme_font_size("font_size")
				var skin := button.get_theme_stylebox("normal")
				var available := button.size.x-skin.get_content_margin(SIDE_LEFT)-skin.get_content_margin(SIDE_RIGHT)
				var longest := 0.0
				for word: String in button.text.split(" "): longest = maxf(longest,font.get_string_size(word,HORIZONTAL_ALIGNMENT_LEFT,-1,font_size).x)
				var headers: Array[Dictionary] = []
				for header: Control in app.menu.headers:
					if header is Label: headers.append({"text":header.text,"font_size":header.get_theme_font_size("font_size"),"rect":rect(header)})
					else: headers.append({"text":header.name,"font_size":0,"rect":rect(header)})
				var footer: Array[Dictionary] = []
				for child: Control in app.menu.footer.get_children(): footer.append({"name":child.name,"text":child.text if child is Button else "","rect":rect(child),"minimum_width":child.get_combined_minimum_size().x})
				rows.append({"viewport":[viewport.x,viewport.y],"scale":scale,"unit":app.menu.ui.unit,"track":Catalog.tracks[index].id,"frame":rect(app.menu.frame),"body":rect(app.menu.body),"list":rect(app.events.track_list),"detail":rect(app.events.detail_scroll),"headers":headers,"footer_panel":rect(app.menu.footer.get_parent()),"footer":footer,"name_font":font_size,"name_button":rect(button),"name_available_width":available,"longest_word_width":longest})
	app.queue_free()
	await create_timer(.5).timeout
	assert(AppData.remove_tree(ProjectSettings.globalize_path(folder))==OK)
	var file := FileAccess.open("/tmp/pacific-density-before-native.json",FileAccess.WRITE)
	file.store_string(JSON.stringify({"cases":rows},"\t")+"\n")
	file.close()
	print("CURRENT_UI_DENSITY_MEASUREMENTS_COMPLETE cases=",rows.size())
	quit()
