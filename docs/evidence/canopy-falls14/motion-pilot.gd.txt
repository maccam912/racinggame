extends SceneTree
## Native pixels prove animated geometry/water moves and Reduce Motion freezes it.
func _initialize() -> void: call_deferred("run")

func capture(path: String) -> Image:
	for frame in range(8): await process_frame
	await RenderingServer.frame_post_draw
	var result := root.get_texture().get_image()
	assert(result.save_png(path)==OK)
	return result

func changed_samples(a: Image, b: Image, tolerance: float = .015) -> int:
	assert(a.get_size()==b.get_size())
	var changed := 0
	for y in range(0,a.get_height(),2):
		for x in range(0,a.get_width(),2):
			var first := a.get_pixel(x,y)
			var last := b.get_pixel(x,y)
			if maxf(absf(first.r-last.r),maxf(absf(first.g-last.g),absf(first.b-last.b)))>tolerance: changed += 1
	return changed

func compare_motion(world: TrackWorld, prefix: String, output: String, tolerance: float = .015) -> Dictionary:
	var settings := SettingsCatalog.defaults(false)
	settings.frame_cap = 60
	settings.vsync = false
	settings.reduce_motion = false
	SettingsRuntime.apply(world,settings)
	var first := await capture(output+"/"+prefix+"-moving-before.png")
	for frame in range(45): await process_frame
	var second := await capture(output+"/"+prefix+"-moving-after.png")
	for frame in range(37): await process_frame
	var third := await capture(output+"/"+prefix+"-moving-later.png")
	var after_count := changed_samples(first,second,tolerance)
	var later_count := changed_samples(first,third,tolerance)
	var moving := maxi(after_count,later_count)
	assert(moving>30,"real rendered motion must be visible: "+prefix)
	settings.reduce_motion = true
	SettingsRuntime.apply(world,settings)
	first = await capture(output+"/"+prefix+"-reduced-before.png")
	for frame in range(45): await process_frame
	second = await capture(output+"/"+prefix+"-reduced-after.png")
	for frame in range(37): await process_frame
	third = await capture(output+"/"+prefix+"-reduced-later.png")
	var reduced := maxi(changed_samples(first,second,tolerance),changed_samples(first,third,tolerance))
	assert(reduced==0,"Reduce Motion freezes all motion pixels: "+prefix)
	print("SET_PIECE_NATIVE_MOTION ",prefix," moving=",moving," reduced=",reduced)
	return {"moving_changed_samples":moving,"moving_after_changed_samples":after_count,"moving_later_changed_samples":later_count,"best_moving_frame":"after" if after_count>=later_count else "later","reduced_changed_samples":reduced,"channel_tolerance":tolerance}

func run() -> void:
	assert(DisplayServer.get_name()!="headless")
	Catalog.load_all()
	root.size = Vector2i(960,540)
	root.content_scale_mode = Window.CONTENT_SCALE_MODE_DISABLED
	DisplayServer.window_move_to_foreground()
	var output := "res://docs/evidence/canopy-falls14/motion"
	DirAccess.make_dir_recursive_absolute(output)
	var reports: Array[Dictionary] = []
	for id: String in ["cloudforest_climb"]:
		var track: TrackDefinition = Catalog.tracks.filter(func(item: TrackDefinition) -> bool: return item.id==id)[0]
		var world := TrackWorld.new()
		world.build(track)
		root.add_child(world)
		var hero := world.find_children("HeroSetPiece_*","Node3D",false,false)[0] as Node3D
		var camera := Camera3D.new()
		camera.current = true
		camera.far = 3000
		camera.fov = 60
		world.add_child(camera)
		camera.global_position = hero.to_global(Vector3(40,88,75) if id=="oasis_citadel" else Vector3(55,42,110))
		camera.look_at(hero.to_global(Vector3(0,60,0) if id=="oasis_citadel" else Vector3(0,34,0)),Vector3.UP)
		var result := await compare_motion(world,id,output)
		result["track"] = id
		reports.append(result)
		world.queue_free()
		for frame in range(4): await process_frame
	var water_reports: Array[Dictionary] = []
	for id: String in []:
		var track: TrackDefinition = Catalog.tracks.filter(func(item: TrackDefinition) -> bool: return item.id==id)[0]
		var world := TrackWorld.new()
		world.build(track)
		root.add_child(world)
		var camera := Camera3D.new()
		camera.current = true
		camera.far = 3000
		camera.fov = 60
		world.add_child(camera)
		if id=="palm_bay":
			var s := CourseArt.original_s(track,.45)
			camera.global_position = track.pose(s,85).origin+Vector3.UP*18
			var target := track.pose(s,280).origin
			target.y = float(CourseArt.terrain_info[id].sea_level)
			camera.look_at(target,Vector3.UP)
		else:
			var pool := world.find_children("WaterfallPool","MeshInstance3D",true,false)[0] as MeshInstance3D
			camera.global_position = pool.global_position+Vector3(12,11,18)
			camera.look_at(pool.global_position+Vector3.UP*.4,Vector3.UP)
		var result := await compare_motion(world,id+"-water",output,.003)
		result["track"] = id
		result["kind"] = "ocean" if id=="palm_bay" else "waterfall_pool"
		water_reports.append(result)
		world.queue_free()
		for frame in range(4): await process_frame
	var file := FileAccess.open(output+"/native.json",FileAccess.WRITE)
	file.store_string(JSON.stringify({"passed":true,"renderer":RenderingServer.get_current_rendering_method(),"resolution":[960,540],"content_hash":Compatibility.content_hash(),"shader_sha256":FileAccess.get_sha256("res://assets/exported/shaders/set_piece_motion.gdshader"),"water_shader_sha256":FileAccess.get_sha256("res://assets/exported/shaders/water.gdshader"),"source_sha256":FileAccess.get_sha256("res://game/world/expansion_set_pieces.gd"),"ambient_source_sha256":FileAccess.get_sha256("res://game/world/expansion_ambient_motion.gd"),"cases":reports,"water_cases":water_reports,"boundary":"Native macOS fixed-camera pixels, every second pixel. Three samples separated by 45 then 37 frames (plus eight settling frames per capture) avoid a periodic sway returning to the same pose. Max of two comparisons, 0.015 channel tolerance for landmarks and 0.003 for subtle water normals. Reduce Motion must yield zero changed samples in both comparisons, including ocean and waterfall pools. Not APK/physical-device or frame-time acceptance."},"\t")+"\n")
	file.close()
	print("NATIVE_CLOUD_FOREST_WATERFALL_AND_REDUCED_MOTION_PIXELS_PASS")
	quit()
